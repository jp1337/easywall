**Goal:** /blocked polls with `after=<seq>` and gets 204 or only the new rows, marks them, can be paused, and says when and why it is paused.

```go
// handleBlockedRows is the live tail: polled every five seconds with after=<seq>,
// the highest Seq the page holds. It answers with the rows newer than that and
// htmx puts them on top (HX-Reswap: afterbegin), or with 204 when there are
// none, so an idle tail costs no body at all. With no after — the page had no
// rows — or with a ring that went backwards (a core restarted without its
// spill file numbers from 1 again), it answers with the whole view. A
// separate request from handleBlocked's, so it asks the core the same two
// questions again rather than inheriting an answer from a page load that may
// be minutes old.
func (s *Server) handleBlockedRows(w http.ResponseWriter, r *http.Request) {
	q := r.URL.Query()
	f, _ := blockedFilter(q) // an unreadable filter here renders as no filter, exactly like the page
	after, _ := strconv.ParseUint(q.Get("after"), 10, 64)
	// The answer is the state of the ring at this moment; nothing between the
	// browser and this process may hand an older one back.
	w.Header().Set("Cache-Control", "no-store")
	rows := blockedRows{Entries: []shared.PacketLogEntry{}, Query: filterQuery(f), Filtered: !f.IsZero()}
	swap := "innerHTML"
	if res, err := s.client.GetPacketLog(f); err == nil {
		rows.Entries = res.Entries
		rows.Listening = res.Listening
		rows.Since = res.Since
		if newer, ok := entriesAfter(res.Entries, after); ok {
			if len(newer) == 0 {
				w.WriteHeader(http.StatusNoContent)
				return
			}
			rows.Entries, swap = newer, "afterbegin"
		}
		s.rulesNow(&rows)
		s.feedNames(r, &rows)
	} else {
		slog.Debug("could not get the packet log for the live tail", "error", err)
		rows.CoreErr = err.Error()
	}
	if opts, err := s.client.GetOptions(); err == nil {
		rows.Logging = opts.LogsAnything()
		rows.LoggingKnown = true
	}
	w.Header().Set("HX-Reswap", swap)
	s.renderPartial(w, r, "blocked_rows", rows)
}

// entriesAfter returns the entries newer than after, newest first as the core
// sends them, and whether a delta is meaningful at all: not without an after,
// and not when the newest entry is older than after — the ring was renumbered
// and the page's rows are no longer comparable.
func entriesAfter(entries []shared.PacketLogEntry, after uint64) ([]shared.PacketLogEntry, bool) {
	if after == 0 || (len(entries) > 0 && entries[0].Seq < after) {
		return nil, false
	}
	i := 0
	for i < len(entries) && entries[i].Seq > after {
		i++
	}
	return entries[:i], true
}
```

```js
/* ── Blocked traffic live tail ────────────────────────────────────────────
   Every five seconds the tail asks for the rows newer than the newest it
   holds (after=<seq>); the server answers 204 when there are none, or with
   only the new rows and HX-Reswap: afterbegin. The page keeps at most
   data-limit rows, marks the latest batch (.pkt-new) and announces its size.
   The poll is skipped while the operator paused it, or while the pointer or
   the keyboard focus is in the table — an insert at the top would slide the
   row under the cursor — and the state line says which of the three it is.
   Ceiling: a screen reader's browse cursor does not move focus, so it is not
   seen here. */
function initBlockedTail() {
  const rows = document.getElementById('blocked-rows');
  const state = document.getElementById('blocked-live-state');
  const toggle = document.getElementById('blocked-live-toggle');
  const announce = document.getElementById('blocked-live-new');
  if (!rows || !state || !toggle) return;
  let paused = false;

  const why = () => paused ? 'paused'
    : rows.matches(':hover') ? 'pointer'
    : rows.contains(document.activeElement) ? 'focus' : 'on';
  const show = () => { state.textContent = state.dataset[why()]; };
  const newest = () => {
    const first = rows.querySelector('tr[data-seq]');
    return first ? first.dataset.seq : '';
  };

  toggle.addEventListener('click', () => {
    paused = !paused;
    toggle.setAttribute('aria-pressed', String(paused));
    toggle.textContent = paused ? toggle.dataset.resume : toggle.dataset.pause;
    show();
  });
  for (const ev of ['pointerenter', 'pointerleave', 'focusin', 'focusout']) {
    // focusout fires before the new focus lands; read the state after it.
    rows.addEventListener(ev, () => setTimeout(show, 0));
  }

  document.body.addEventListener('htmx:beforeRequest', e => {
    if (e.detail.elt !== rows) return;
    if (why() !== 'on') e.preventDefault();
  });
  document.body.addEventListener('htmx:configRequest', e => {
    if (e.detail.elt !== rows) return;
    const seq = newest();
    if (seq) e.detail.parameters.after = seq;
  });
  document.body.addEventListener('htmx:afterSwap', e => {
    if (e.detail.elt !== rows || e.detail.xhr.status !== 200) return;
    rows.querySelectorAll('tr.pkt-new').forEach(tr => tr.classList.remove('pkt-new'));
    if (e.detail.xhr.getResponseHeader('HX-Reswap') !== 'afterbegin') {
      rows.querySelectorAll('tr[data-seq]').forEach(tr => { tr.dataset.seen = '1'; });
      return;
    }
    let added = 0;
    for (const tr of rows.querySelectorAll('tr[data-seq]')) {
      if (tr.dataset.seen) break;
      tr.dataset.seen = '1';
      tr.classList.add('pkt-new');
      added++;
    }
    const limit = Number(rows.dataset.limit) || 200;
    [...rows.querySelectorAll('tr[data-seq]')].slice(limit).forEach(tr => tr.remove());
    if (announce && added) {
      announce.textContent = (added === 1 ? announce.dataset.one : announce.dataset.many).replace('{n}', added);
    }
  });
  rows.querySelectorAll('tr[data-seq]').forEach(tr => { tr.dataset.seen = '1'; });
  show();
}
```

```js
/**
 * /blocked's live tail (2.21.1 F3, F12; 2.25.2). It must not move what the
 * operator is pointing at or destroy what has keyboard focus, must say when
 * and why it is paused, must ask only for what is new, and must stop asking
 * while the Pause button is pressed. The demo refuses a packet every 7 s, so
 * 12 s is at least one poll with new rows either way.
 */
async function checkBlockedTailHoldsStill(page) {
  const rowsSeen = [];
  const onResponse = async res => {
    if (!res.url().includes('/blocked/rows')) return;
    rowsSeen.push({ url: res.url(), status: res.status(), swap: res.headers()['hx-reswap'] || '',
                    bytes: (await res.body().catch(() => Buffer.alloc(0))).length });
  };
  page.on('response', onResponse);
  try {
    await page.goto(`${BASE}/blocked`, { waitUntil: 'networkidle' });
    if (await page.locator('#blocked-rows details').count() === 0) {
      fail('blocked tail', 'no rows to observe — the demo produced no refused packets');
      return;
    }
    const firstSeq = () => page.$eval('#blocked-rows details', d => d.id);
    const stateText = () => page.$eval('#blocked-live-state', s => s.textContent.trim());
    const liveText = await page.$eval('#blocked-live-state', s => s.dataset.on);

    await page.hover('#blocked-rows tr:first-child td:first-child');
    const underPointer = await firstSeq();
    await page.waitForTimeout(12000);
    if (await firstSeq() !== underPointer) fail('blocked tail', 'rows moved while the pointer was on the table');
    if (await stateText() === liveText) fail('blocked tail', 'the state line says live while the pointer pauses the tail');

    rowsSeen.length = 0;
    await page.mouse.move(5, 5);
    await page.waitForTimeout(12000);
    if (await firstSeq() === underPointer) fail('blocked tail', 'the tail did not resume after the pointer left the table');
    if (await stateText() !== liveText) fail('blocked tail', `the state line reads "${await stateText()}" while live`);
    const asked = rowsSeen.filter(r => r.status === 200 || r.status === 204);
    if (asked.length === 0 || !asked.every(r => /[?&]after=\d+/.test(r.url))) {
      fail('blocked tail', `a poll did not carry after=<seq>: ${asked.map(r => r.url).join(', ')}`);
    }
    if (asked.some(r => r.status === 200 && r.swap !== 'afterbegin')) {
      fail('blocked tail', 'a poll with new rows answered with the whole view');
    }
    if (asked.some(r => r.status === 204 && r.bytes !== 0)) fail('blocked tail', 'a poll with nothing new carried a body');
    const marked = await page.locator('#blocked-rows tr.pkt-new').count();
    if (marked === 0) fail('blocked tail', 'the newest batch is not marked');
    const limit = await page.$eval('#blocked-rows', t => Number(t.dataset.limit));
    if (await page.locator('#blocked-rows tr[data-seq]').count() > limit) fail('blocked tail', `more than ${limit} rows on the page`);

    await page.click('#blocked-live-toggle');
    await page.mouse.move(5, 5);
    if (await page.$eval('#blocked-live-toggle', b => b.getAttribute('aria-pressed')) !== 'true') {
      fail('blocked tail', 'the Pause button is not pressed after a click');
    }
    rowsSeen.length = 0;
    const paused = await firstSeq();
    await page.waitForTimeout(12000);
    if (rowsSeen.length > 0) fail('blocked tail', `${rowsSeen.length} polls were sent while paused`);
    if (await firstSeq() !== paused) fail('blocked tail', 'rows arrived while paused');
    await page.click('#blocked-live-toggle');
    await page.mouse.move(5, 5);
    await page.waitForTimeout(7000);
    if (await firstSeq() === paused) fail('blocked tail', 'Resume did not catch up');

    if (await page.locator('#blocked-rows .pkt-actions .btn').count() === 0) {
      fail('blocked tail', 'no row action to focus — the demo produced no refused packets');
      return;
    }
    await page.focus('#blocked-rows .pkt-actions .btn');
    const label = await page.evaluate(() => document.activeElement.textContent.trim());
    await page.waitForTimeout(12000);
    const still = await page.evaluate(() => {
      const a = document.activeElement;
      return a && a.isConnected && !!a.closest('#blocked-rows') ? a.textContent.trim() : null;
    });
    if (still !== label) {
      fail('blocked tail', `keyboard focus on "${label}" was lost to ${still === null ? 'the page' : `"${still}"`}`);
      return;
    }
    console.log('  ok   blocked tail asks only for what is new, holds still under pointer, focus and Pause, says so, resumes after');
  } finally {
    page.off('response', onResponse);
  }
}
```
